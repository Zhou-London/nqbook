function(nqhpc_define_binary_path target visibility macro binary)
    target_compile_definitions(${target} ${visibility} ${macro}="$<TARGET_FILE:${binary}>")
    add_dependencies(${target} ${binary})
endfunction()

function(nqhpc_add_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE nqhpc_tests)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
